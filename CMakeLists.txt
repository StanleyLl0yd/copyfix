cmake_minimum_required(VERSION 3.21)
cmake_policy(SET CMP0091 NEW)
project(CopyFix LANGUAGES CXX VERSION 1.0.0)
configure_file(resources/version.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/copyfix_version.h" @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(copyfix_core STATIC src/text_engine.cpp src/diff_spans.cpp)
target_include_directories(copyfix_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")

include(CTest)
if(BUILD_TESTING)
  add_executable(copyfix_tests tests/core_tests.cpp)
  target_link_libraries(copyfix_tests PRIVATE copyfix_core)
  add_test(NAME core COMMAND copyfix_tests)
  add_executable(copyfix_diff_tests tests/diff_tests.cpp)
  target_link_libraries(copyfix_diff_tests PRIVATE copyfix_core)
  add_test(NAME preview_diff COMMAND copyfix_diff_tests)
  add_executable(copyfix_perf_tests tests/perf_tests.cpp)
  target_link_libraries(copyfix_perf_tests PRIVATE copyfix_core)
  add_test(NAME perf_integrity COMMAND copyfix_perf_tests)
endif()

if(WIN32)
  enable_language(RC)
  add_executable(CopyFix WIN32 src/winmain.cpp resources/copyfix.rc)
  target_include_directories(CopyFix PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/resources" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(CopyFix PRIVATE UNICODE _UNICODE)
  target_link_libraries(CopyFix PRIVATE copyfix_core user32 shell32 gdi32 comctl32 advapi32)
  if(MSVC)
    target_link_options(CopyFix PRIVATE "/MANIFESTDEPENDENCY:type='win32' name='Microsoft.Windows.Common-Controls' version='6.0.0.0' processorArchitecture='*' publicKeyToken='6595b64144ccf1df' language='*'")
  endif()
  if(BUILD_TESTING)
    # Separate native smoke test binary. Never shipped with the portable EXE.
    add_executable(copyfix_preview_smoke tests/preview_smoke.cpp)
    target_compile_definitions(copyfix_preview_smoke PRIVATE UNICODE _UNICODE)
    target_link_libraries(copyfix_preview_smoke PRIVATE copyfix_core user32 shell32 gdi32 comctl32 advapi32)
    add_test(NAME preview_scroll COMMAND copyfix_preview_smoke)
    # Synthetic CF_UNICODETEXT integration, Undo, stale sequence and rich-format guard.
    add_executable(copyfix_clipboard_smoke tests/clipboard_smoke.cpp)
    target_compile_definitions(copyfix_clipboard_smoke PRIVATE UNICODE _UNICODE)
    target_link_libraries(copyfix_clipboard_smoke PRIVATE copyfix_core user32 shell32 gdi32 comctl32 advapi32)
    add_test(NAME clipboard_contract COMMAND copyfix_clipboard_smoke)
    # Per-user registry and transaction-safe global hotkey registration.
    add_executable(copyfix_settings_smoke tests/settings_smoke.cpp)
    target_compile_definitions(copyfix_settings_smoke PRIVATE UNICODE _UNICODE)
    target_link_libraries(copyfix_settings_smoke PRIVATE copyfix_core user32 shell32 gdi32 comctl32 advapi32)
    add_test(NAME settings_registry COMMAND copyfix_settings_smoke)
    # Verify the portable EXE's Windows VersionInfo metadata matches the build version.
    add_executable(copyfix_version_smoke tests/version_smoke.cpp)
    target_include_directories(copyfix_version_smoke PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    target_link_libraries(copyfix_version_smoke PRIVATE version)
    add_test(NAME version_resources COMMAND copyfix_version_smoke "$<TARGET_FILE:CopyFix>")
    foreach(target copyfix_preview_smoke copyfix_clipboard_smoke copyfix_settings_smoke)
      target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    endforeach()
    # Inspect the actual embedded Windows icon resource without shipping test binaries.
    add_executable(copyfix_icon_smoke tests/icon_smoke.cpp)
    target_compile_definitions(copyfix_icon_smoke PRIVATE UNICODE _UNICODE)
    target_include_directories(copyfix_icon_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(copyfix_icon_smoke PRIVATE user32)
    add_test(NAME icon_resources COMMAND copyfix_icon_smoke "$<TARGET_FILE:CopyFix>" "${CMAKE_CURRENT_SOURCE_DIR}/artwork/tray-16x16.png")
  endif()
  set_target_properties(CopyFix PROPERTIES OUTPUT_NAME CopyFix)
  foreach(target copyfix_core CopyFix copyfix_tests copyfix_perf_tests copyfix_diff_tests copyfix_preview_smoke copyfix_clipboard_smoke copyfix_settings_smoke copyfix_icon_smoke copyfix_version_smoke)
    if(TARGET ${target})
      set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /GS /sdl /guard:cf $<$<CONFIG:Release>:/analyze>)
        target_compile_options(${target} PRIVATE $<$<CONFIG:Release>:/O1> $<$<CONFIG:Release>:/GL> $<$<CONFIG:Release>:/Gy> $<$<CONFIG:Release>:/GF>)
      endif()
    endif()
  endforeach()
  if(MSVC)
    target_link_options(CopyFix PRIVATE /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /GUARD:CF $<$<CONFIG:Release>:/LTCG> $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF> $<$<CONFIG:Release>:/INCREMENTAL:NO>)
  endif()
endif()
